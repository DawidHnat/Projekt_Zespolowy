\section*{1. Kanały komunikacji i czas odpowiedzi}

\section*{2. Dostępność członków i harmonogram spotkań}
\begin{itemize}
    \item \textbf{Spotkania cykliczne:} Minimum 1 spotkanie w tygodniu. Dokładny termin spotkania ustalany jest wspólnie w każdy poniedziałek.
    \item \textbf{Procedura zastoju:} W przypadku braku aktywności w projekcie (np. braku aktualizacji postępów przez 3 dni), Lider zespołu ma obowiązek wyznaczyć dodatkowe, obowiązkowe spotkanie grupowe w uzgodnionym przez wszystkich terminie.
\end{itemize}

\section*{3. Sposób podejmowania decyzji}
\begin{itemize}
    \item Decyzje projektowe podejmowane są wspólnie na drodze głosowania. Zawsze wygrywa opcja, która zdobędzie większość głosów.
    \item W przypadku remisu ostateczną decyzję podejmuje Lider zespołu.
\end{itemize}

\section*{4. Postępowanie w przypadku niedowiezienia zadania}
\begin{itemize}
    \item \textbf{Błąd indywidualny:} Jeśli opóźnienie wynika z zaniedbania lub błędu konkretnej osoby, ponosi ona odpowiedzialność za nadrobienie zaległości. Lider może wyznaczyć innych członków zespołu, aby wsparli tę osobę w ukończeniu pracy.
    \item \textbf{Problem zespołowy:} Jeśli problem wynika z trudności technicznych lub dotyczy całego zespołu, zwoływana jest wspólna narada w celu przeanalizowania sytuacji i wdrożenia kroków naprawczych.
\end{itemize}
